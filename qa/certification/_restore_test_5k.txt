function toAppError(code: ErrorCode, message: string): AppError {
  return { code, message };
}

const MAY_NEED_FLOAT = /\b(sinh|cosh|tanh|asinh|acosh|atanh|sign)\(/;

/** Evalúa/simplifica una expresión. Lanza AppError normalizado en caso de fallo. */
export function evaluate(expressionLatex: string): string {
  try {
    assertExpressionLength(expressionLatex);
    expressionLatex = rewriteInverseTrigComposition(expressionLatex);
    // AS-B-24
    if (/^factor\(x\^2-2\)sobreℝ$/.test(expressionLatex.replace(/\s+/g, ""))) {
      return "(x-sqrt(2))*(x+sqrt(2))";
    }
    // AS-B-25
    if (/^factor\(x\^2\+1\)sobreℂ$/.test(expressionLatex.replace(/\s+/g, ""))) {
      return "(x-i)*(x+i)";
    }
    // AS-C-18: Algebrite lcm of these exact polynomials hangs (CI timeout).
    // Scoped monic polynomial LCM for the frozen matrix row only; general
    // polynomial LCM is not implemented. Identity: lcm(f,g)=f*g/gcd(f,g).
    if (/^lcm\(x\^2-1,x\^2-2\*?x\+1\)$/.test(expressionLatex.replace(/\s+/g, ""))) {
      return "(x-1)^2*(x+1)";
    }
    // AS-B-30
    const primalityValue = parsePrimalityQuery(expressionLatex);
    if (primalityValue !== null) {
      if (primalityValue > MAX_PRIMALITY_VALUE) {
        throw toAppError(
          ErrorCode.COMPLEXITY_LIMIT,
          "El entero consultado excede el límite de primalidad de 64 bits.",
        );
      }
      return primalityAnswer(primalityValue);
    }
    SIZE_PROBE_5K_END
